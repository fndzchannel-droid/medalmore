if ( ! defined( 'ABSPATH' ) ) { exit; }

/** Fetch the hub JSON from GitHub raw, cached 5 minutes via a transient. */
if ( ! function_exists( 'mm_fifa_hub_data' ) ) {
    function mm_fifa_hub_data() {
        $cached = get_transient( 'mm_fifa_hub' );
        if ( $cached !== false ) { return $cached; }

        $url  = 'https://raw.githubusercontent.com/fndzchannel-droid/medalmore/main/fifa-hub.json';
        $resp = wp_remote_get( $url, array( 'timeout' => 8 ) );
        if ( is_wp_error( $resp ) || wp_remote_retrieve_response_code( $resp ) !== 200 ) {
            return get_option( 'mm_fifa_hub_last', array() ); // fall back to last good copy
        }
        $data = json_decode( wp_remote_retrieve_body( $resp ), true );
        if ( ! is_array( $data ) ) {
            return get_option( 'mm_fifa_hub_last', array() );
        }
        set_transient( 'mm_fifa_hub', $data, 5 * MINUTE_IN_SECONDS );
        update_option( 'mm_fifa_hub_last', $data ); // durable fallback
        return $data;
    }
}

$hub      = mm_fifa_hub_data();
$results  = isset( $hub['results'] )  ? $hub['results']  : array();
$fixtures = isset( $hub['fixtures'] ) ? $hub['fixtures'] : array();
$schedule = isset( $hub['schedule'] ) ? $hub['schedule'] : array();
$through  = isset( $hub['through'] )  ? $hub['through']  : array();
$champion = isset( $hub['champion'] ) ? trim( $hub['champion'] ) : '';
$stage    = isset( $hub['stage'] )    ? $hub['stage']    : 'Round of 32';
$updated  = isset( $hub['updated'] )  ? $hub['updated']  : '';
